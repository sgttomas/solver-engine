   from typing import Any
   from uuid import UUID
   from sqlalchemy.ext.asyncio import AsyncSession

   from domain.state import StepName, STEP_NUMBERS
   from infrastructure.db.models import TraceabilityLink
   from infrastructure.db.repositories import TraceabilityLinkRepository


   class TraceabilityService:
       """Extract and persist traceability links from step packages."""

       def __init__(self, session: AsyncSession) -> None:
           self._session = session
           self._trace_repo = TraceabilityLinkRepository(session)

       async def extract_and_persist_links(
           self,
           workflow_id: UUID,
           step_name: StepName,
           package_content: dict[str, Any],
       ) -> list[TraceabilityLink]:
           """Extract trace links from package and persist idempotently.

           Args:
               workflow_id: Database workflow UUID.
               step_name: Step name (REQUIREMENTS or OBJECTIVES).
               package_content: Full package dict with metadata.

           Returns:
               List of created TraceabilityLink records.
           """
           step_number = STEP_NUMBERS[step_name]

           # Step 1 has no forward links (it's the source)
           if step_name == StepName.PROBLEM_DEFINITION:
               return []

           links = []

           if step_name == StepName.REQUIREMENTS:
               links = self._extract_step2_links(step_number, package_content)
           elif step_name == StepName.OBJECTIVES:
               links = self._extract_step3_links(step_number, package_content)

           # Persist idempotently
           created = []
           for link_data in links:
               existing = await self._trace_repo.find_by_unique(
                   workflow_id=workflow_id,
                   from_id=link_data["from_id"],
                   to_id=link_data["to_id"],
                   link_type=link_data["link_type"],
               )
               if existing:
                   continue

               link = TraceabilityLink(
                   workflow_id=workflow_id,
                   from_step=link_data["from_step"],
                   from_type=link_data["from_type"],
                   from_id=link_data["from_id"],
                   to_step=link_data["to_step"],
                   to_type=link_data["to_type"],
                   to_id=link_data["to_id"],
                   link_type=link_data["link_type"],
                   consolidated=link_data.get("consolidated", False),
               )
               self._session.add(link)
               created.append(link)

           if created:
               await self._session.flush()

           return created

       def _extract_step2_links(
           self,
           step_number: int,
           package: dict[str, Any],

           """Extract Step 1 → Step 2 links from RequirementsPackage."""
           links = []

           for req in package.get("requirements", []):
               req_id = req.get("id")
               if not req_id:
                   continue

               req_type = self._get_type_from_id(req_id)

               # source.id → primary derivation
               source = req.get("source", {})
               if source.get("id"):
                   links.append({
                       "from_step": 1,
                       "from_type": source.get("type", "unknown"),
                       "from_id": source["id"],
                       "to_step": step_number,
                       "to_type": req_type,
                       "to_id": req_id,
                       "link_type": "derives",
                   })

               # trace object → additional links
               trace = req.get("trace", {})

               for sh_id in trace.get("stakeholders", []):
                   links.append({
                       "from_step": 1,
                       "from_type": "stakeholder",
                       "from_id": sh_id,
                       "to_step": step_number,
                       "to_type": req_type,
                       "to_id": req_id,
                       "link_type": "traces_to",
                   })

               for const_id in trace.get("constraints", []):
                   const_type = "hard_constraint" if const_id.startswith("HC-")
  else "soft_constraint"
                   links.append({
                       "from_step": 1,

                       "from_id": const_id,
                       "to_step": step_number,
                       "to_type": req_type,
                       "to_id": req_id,

       def _get_type_from_id(self, element_id: str) -> str:

           if not element_id or "-" not in element_id:

               "CR": "constraint_requirement",
               "IR": "interface_requirement",
               "CAP": "capability_objective",
               "QUAL": "quality_objective",
               "COMP": "compliance_objective",
               "INTF": "interface_objective",
           }
           return type_map.get(prefix, "unknown")

   #!/usr/bin/env python3
   """
   Gate B Verification: Traceability Links

   Usage: python tools/verify_traces.py --workflow <workflow_id>
   """

   import argparse
   import asyncio
   import sys

   from sqlalchemy import select, func
   from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession
   from sqlalchemy.orm import sessionmaker

   from infrastructure.db.models import Artifact, TraceabilityLink
   from infrastructure.db.models.enums import PassType


   async def verify_traces(workflow_id: str, database_url: str) -> bool:
       """Verify traceability links exist for a workflow."""
       engine = create_async_engine(database_url)
       async_session = sessionmaker(engine, class_=AsyncSession,
  expire_on_commit=False)

       async with async_session() as session:
           # Get workflow UUID from external ID
           from infrastructure.db.models import Workflow
           stmt = select(Workflow).where(Workflow.workflow_id == workflow_id)
           result = await session.execute(stmt)
           workflow = result.scalar_one_or_none()

           if not workflow:
               print(f"ERROR: Workflow not found: {workflow_id}")
               return False

           # Count trace links by step transition
           stmt = (
               select(
                   TraceabilityLink.from_step,
                   TraceabilityLink.to_step,
                   func.count(TraceabilityLink.id).label("count"),
               )
               .where(TraceabilityLink.workflow_id == workflow.id)
               .group_by(TraceabilityLink.from_step, TraceabilityLink.to_step)
           )
           result = await session.execute(stmt)
           counts = {(row.from_step, row.to_step): row.count for row in result}

           print(f"Workflow: {workflow_id}")
           print(f"Step 1 → Step 2 links: {counts.get((1, 2), 0)}")
           print(f"Step 2 → Step 3 links: {counts.get((2, 3), 0)}")
           print(f"Step 1 → Step 3 links: {counts.get((1, 3), 0)}")

           # Gate B requirement: Step 2 and 3 must have incoming traces
           step2_has_traces = counts.get((1, 2), 0) > 0
           step3_has_traces = counts.get((2, 3), 0) > 0

           if step2_has_traces and step3_has_traces:
               print("✓ Gate B: Traceability links valid")
               return True
           else:
               print("✗ Gate B: Missing traceability links")
               if not step2_has_traces:
                   print("  - Step 2 missing traces from Step 1")
               if not step3_has_traces:
                   print("  - Step 3 missing traces from Step 2")
               return False


   def main():
       parser = argparse.ArgumentParser(description="Verify traceability links
  (Gate B)")
       parser.add_argument("--workflow", required=True, help="Workflow ID to
  verify")
       parser.add_argument("--database-url", default=None, help="Database URL
  (or use DATABASE_URL env)")
       args = parser.parse_args()

       import os
       database_url = args.database_url or os.environ.get("DATABASE_URL")
       if not database_url:
           print("ERROR: DATABASE_URL not set")
           sys.exit(1)

       success = asyncio.run(verify_traces(args.workflow, database_url))
       sys.exit(0 if success else 1)


   if __name__ == "__main__":
       main()
